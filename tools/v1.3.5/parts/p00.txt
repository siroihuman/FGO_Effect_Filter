  function normalizeText(value) {
    return String(value || '')
      .replace(/\u00a0/g, ' ')
      .replace(/[\t\r]+/g, ' ')
      .replace(/ +/g, ' ')
      .replace(/\n{3,}/g, '\n\n')
      .trim();
  }

  function escapeHtml(value) {
    return String(value ?? '').replace(/[&<>'"]/g, ch => ({
      '&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'
    }[ch]));
  }

  function headingLevel(el) {
    const m = /^H([1-6])$/.exec(el?.tagName || '');
    return m ? Number(m[1]) : 99;
  }

  function findHeading(doc, exactText) {
    return [...doc.querySelectorAll('h1,h2,h3,h4,h5,h6')]
      .find(h => normalizeText(h.textContent).replace(/[▼▽]/g,'') === exactText) || null;
  }

  function collectSectionText(heading) {
    if (!heading) return '';
    const level = headingLevel(heading);
    const chunks = [];
    let node = heading.nextElementSibling;
    while (node) {
      if (/^H[1-6]$/.test(node.tagName) && headingLevel(node) <= level) break;
      chunks.push(node.innerText || node.textContent || '');
      node = node.nextElementSibling;
    }
    return normalizeText(chunks.join('\n'));
  }
